class CSRBuilder:


    def __init__(self):

        self.graph = CSRGraph()

        self.context_stack = []


    def add_node(
        self,
        node
    ):

        self.graph.add(node)

        return node

    def add_expression_node(
        self,
        node
    ):

        self.graph.add(node)

        return node

    def identifier(
        self,
        name
    ):

        node = IdentifierNode(
            name
        )

        return self.add_expression_node(
            node
        )

    def literal(
        self,
        value
    ):

        node = LiteralNode(
            value
        )

        return self.add_expression_node(
            node
        )

    def binary(
        self,
        operator,
        left,
        right
    ):

        node = BinaryOperationNode(
            operator
        )


        self.add_expression_node(
            node
        )


        self.graph.link(
            node,
            left,
            RelationType.LEFT
        )


        self.graph.link(
            node,
            right,
            RelationType.RIGHT
        )


        return node

    def call(
        self,
        target,
        arguments=None
    ):

        node = CallNode()


        self.add_expression_node(
            node
        )


        self.graph.link(
            node,
            target,
            RelationType.TARGET
        )


        for argument in arguments or []:

            self.graph.link(
                node,
                argument,
                RelationType.ARGUMENT
            )


        return node

    def assignment(
        self,
        left,
        right
    ):

        node = AssignmentNode()

        self.add_node(node)

        self.graph.link(
            node,
            left,
            RelationType.LEFT
        )

        self.graph.link(
            node,
            right,
            RelationType.RIGHT
        )

        self.add_to_current_body(
            node
        )

        return node
    
    def link(
        self,
        source,
        target,
        relation
    ):

        self.graph.link(
            source,
            target,
            relation
        )

    def find_context(self, allowed_types):

        for context in reversed(self.context_stack):

            if context.type in allowed_types:
                return context.node

        return None

    def begin_function(
        self,
        name,
        signature=""
    ):

        function = FunctionNode(
            name,
            signature
        )

        body = BodyNode()

        self.add_node(function)
        self.add_node(body)

        #
        # collega la funzione al File corrente
        #
        parent = self.current_context()

        if parent and parent.type not in [
            "file",
            "class"
        ]:
            parent = self.parent_context()
        if parent is not None:

            self.link(
                parent,
                function,
                RelationType.FUNCTION
            )

        #
        # collega il body
        #
        self.link(
            function,
            body,
            RelationType.BODY
        )

        self.push_context(function)
        self.push_context(body)

        return function

    def add_parameter(
        self,
        name,
        data_type=None
    ):

        parameter = ParameterNode(
            name,
            data_type
        )


        self.add_node(
            parameter
        )


        function = self.parent_context()


        self.link(
            function,
            parameter,
            RelationType.PARAMETER
        )


        return parameter



    def add_return(
        self
    ):

        node = ReturnNode()


        self.add_node(
            node
        )


        body = self.current_context()

        if body is None:
            return node

        self.link(
            body,
            node,
            RelationType.CONTAINS
        )


        return node



    def end_function(self):
        self.pop_context()
        self.pop_context()
    
    def current_context(self):

        if not self.context_stack:
            return None

        return self.context_stack[-1].node
    
    def parent_context(self):

        if len(self.context_stack) < 2:
            return None

        return self.context_stack[-2].node

    def add_to_current_body(self, node):

        parent = self.current_context()

        if parent is None:
            return

        self.link(
            parent,
            node,
            RelationType.CONTAINS
        )
        
    def push_context(

        node
    ):

from collections import defaultdict

from .edge import CSREdge
from .relation import RelationType


class CSRGraph:

    def __init__(self):

        self.nodes = {}

        self.edges = []

        self._outgoing = defaultdict(list)

        self._incoming = defaultdict(list)


    def add(self, node):

        self.nodes[node.id] = node

        return node


    def link(
        self,
        source,
        target,
        relation,
        **attributes
    ):

        edge = CSREdge(
            source.id,
            target.id,
            relation,
            attributes=attributes
        )

        self.edges.append(edge)

        self._outgoing[source.id].append(edge)

        self._incoming[target.id].append(edge)

        return edge


    def children(
        self,
        node,
        relation=None
    ):

        result = []

        for edge in self._outgoing[node.id]:

            if relation and edge.relation != relation:

                continue

            result.append(
                self.nodes[edge.target]
            )

        return result


    def parents(
        self,
        node,
        relation=None
    ):

        result = []

        for edge in self._incoming[node.id]:

            if relation and edge.relation != relation:

                continue

            result.append(
                self.nodes[edge.source]
            )

        return result


    def find(
        self,
        node_type=None,
        name=None
    ):

        result = []

        for node in self.nodes.values():

            if node_type and node.type != node_type:

                continue

            if name and node.name != name:

                continue

            result.append(node)

        return result

import json


class CSRSerializer:


    def serialize(
        self,
        graph
    ):

        return json.dumps(
            {
                "nodes": [
                    self._node(n)
                    for n in graph.nodes.values()
                ],
                "edges": [
                    self._edge(e)
                    for e in graph.edges
                ]
            },
            indent=2
        )



    def _node(
        self,
        node
    ):

        return {
            "id": node.id,
            "type": node.type,
            "name": node.name,
            "attributes": node.attributes
        }



    def _edge(
        self,
        edge
    ):

        return {
            "id": edge.id,
            "source": edge.source,
            "target": edge.target,
            "relation": edge.relation,
            "attributes": edge.attributes
        }
